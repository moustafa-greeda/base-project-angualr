Usage

<!-- section = grid: [cols] 1 | 2 | 3 (mobile always collapses to 1) -->
<app-form-section title="Basic" [cols]="2">

  <!-- default type is text; validation messages are automatic -->
  <app-form-field label="Name" controlName="name" />

  <!-- types: text · number · email · password · date · textarea · select
       search-select · multi-select · radio · checkbox · switch · file -->

  <!-- searchable dropdown; add a `hint` to an option for a second line -->
  <app-form-field label="Manager" controlName="manager" type="search-select"
                  [options]="employees" />

  <!-- same, but the control holds an array of values -->
  <app-form-field label="Skills" controlName="skillSet" type="multi-select"
                  [options]="skills" />
  <app-form-field label="Department" controlName="department" type="select"
                  placeholder="Choose department" [options]="departments" />
  <app-form-field label="Gender" controlName="gender" type="radio" [options]="genders" />
  <app-form-field label="Active" controlName="active" type="switch" />

  <!-- file: control value is a File (or File[] with [multiple]) -->
  <app-form-field label="Contract" controlName="contract" type="file" accept=".pdf" />
  <app-form-field label="Files" controlName="attachments" type="file"
                  [multiple]="true" accept="image/*,.pdf" />

  <!-- span="full" → field takes the whole row -->
  <app-form-field label="Bio" controlName="bio" type="textarea" span="full" />

  <!-- nested groups via dot path — no formGroupName needed -->
  <app-form-field label="City" controlName="address.city" />

  <!-- custom error text (optional) -->
  <app-form-field label="Name" controlName="name"
                  [errorMessages]="{{ '{' }} required: 'الاسم مطلوب' {{ '}' }}" />
</app-form-section>
