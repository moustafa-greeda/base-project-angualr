Icon Only

<!-- square icon-only button — ariaLabel is required for screen readers -->
<app-button icon="bi bi-pencil-square" variant="outline" [iconOnly]="true" ariaLabel="Edit"
            (clickAction)="onEdit(row)" />
<app-button icon="bi bi-trash" variant="dangerOutline" [iconOnly]="true" ariaLabel="Delete"
            (clickAction)="onDelete(row)" />

As Link

<!-- [link] switches the render from <button> to a real <a routerLink> with the same styling.
     A real anchor gives you for free:
       • Ctrl+Click / middle-click  → opens in a new tab
       • right-click → copy link
       • screen readers announce it as a "link"
     No (clickAction) here — navigation IS the action.
     Rule: button DOES something → (clickAction) · button TAKES YOU somewhere → [link] -->

<app-button titleBtn="Go to Cards" icon="bi bi-card-heading" [link]="'/dashboard/cards'" />

<!-- need to run logic BEFORE navigating? use (clickAction) + router.navigate() instead of [link] -->

States

<!-- loading: shows a spinner and disables the button — no double-submit -->
<app-button titleBtn="Save" [loading]="saving()" (clickAction)="save()" />

<!-- in the component: -->
saving = signal(false);
save() {{ '{' }}
  this.saving.set(true);
  this.api.save().subscribe({{ '{' }} next: () => this.saving.set(false) {{ '}' }});
{{ '}' }}

<app-button titleBtn="Disabled" [disabled]="true" />

Full Width & Submit 

<!-- fullWidth: spans the container — login screens and forms -->
<app-button titleBtn="Submit" [fullWidth]="true" />

<!-- inside a form: type="submit" triggers (ngSubmit) — no clickAction needed -->
<app-button titleBtn="Save" type="submit" />
