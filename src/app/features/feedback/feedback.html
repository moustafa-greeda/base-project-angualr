Toasts

<!-- No markup needed — <app-toast /> lives once in app.html.
     Inject the service anywhere and call it: -->

private toast = inject(ToastService);

this.toast.success('The item was saved successfully');
this.toast.error('Could not reach the server');
this.toast.info('3 new requests are waiting');
this.toast.warning('Your session expires soon');

// optional title as a second argument:
this.toast.success('Invoice #1042 was sent', 'Invoice sent');

// toasts auto-dismiss after 4s, or via the × button
