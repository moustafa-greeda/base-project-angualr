1 — Pipe

<!-- simplest form; re-renders on language change -->
{{ '{{' }} 'common.save' | transloco {{ '}}' }}

<!-- as an input -->
<app-button [titleBtn]="'common.save' | transloco" />

2 — Structural directive (scoped, faster)

{{ t('nav.roles') }}

{{ t('nav.permissions') }}

{{ t('roles.stats.users') }}

<!-- one subscription for the whole block — prefer this in big templates -->
<div *transloco="let t">
  {{ '{{' }} t('nav.roles') {{ '}}' }}
  {{ '{{' }} t('nav.permissions') {{ '}}' }}
</div>

3 — Parameters

{{ 'demo.greeting' | transloco: { name: name() } }}

{{ 'demo.itemCount' | transloco: { count: itemCount() } }}

count = {{ itemCount() }} 
// en.json → "greeting": "Hello, {{ '{{' }} name {{ '}}' }}!"
// ar.json → "greeting": "أهلاً يا {{ '{{' }} name {{ '}}' }}!"

{{ '{{' }} 'demo.greeting' | transloco: {{ '{' }} name: name() {{ '}' }} {{ '}}' }}
